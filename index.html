Task: Sum of Remainders

For each number from 1 to n, calculate i % m and return the sum. 

function f(n, m) {
  let res = 0;
  let cycleNumber = Math.floor(n / m);
  if (cycleNumber == 0) {
    res = (n * (n + 1)) / 2;
    return res;
  }

  let temp = n;
  const maxNumInCycle = m - 1;
  while (temp > 0) {
    if (temp >= m) {
      res = res + (maxNumInCycle * (maxNumInCycle + 1)) / 2;
    } else {
      res = res + (temp * (temp + 1)) / 2;
    }

    temp = temp - m;
  }

  return res;
}
